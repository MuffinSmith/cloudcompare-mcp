cmake_minimum_required(VERSION 3.16)
project(camera_guard_contract LANGUAGES CXX)
enable_testing()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
add_executable(camera_guard test_camera_guard.cpp)
target_compile_features(camera_guard PRIVATE cxx_std_14)
target_include_directories(camera_guard PRIVATE ../../../cloudcompare-plugin/qMCPBridge/include)
target_link_libraries(camera_guard PRIVATE Qt${QT_VERSION_MAJOR}::Core)
foreach(case RANGE 0 36)
    add_test(NAME camera_guard_${case} COMMAND camera_guard ${case})
endforeach()

add_executable(registration_policy test_registration_policy.cpp)
target_compile_features(registration_policy PRIVATE cxx_std_14)
target_include_directories(registration_policy PRIVATE ../../../cloudcompare-plugin/qMCPBridge/include)
foreach(case RANGE 0 4)
    add_test(NAME registration_policy_${case} COMMAND registration_policy ${case})
endforeach()
